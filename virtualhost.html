Plantilla virtualhost /etc/apache2/sites-available/proyecto.tudominio.com.conf


<VirtualHost *:80>
    ServerName proyecto.tudominio.com
    DocumentRoot /var/www/proyecto

    <Directory /var/www/proyecto>
        Options -Indexes +FollowSymLinks
        AllowOverride None
        Require all granted
    </Directory>

    ErrorLog ${APACHE_LOG_DIR}/proyecto-error.log
    CustomLog ${APACHE_LOG_DIR}/proyecto-access.log combined
</VirtualHost>
